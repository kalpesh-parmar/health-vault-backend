================================================================================
  SWAGGER TEST GUIDE: FAILED DOCUMENT UPLOAD & MANUAL RECOVERY FLOW
  Endpoint: POST /v1/onboarding-chat
================================================================================

--------------------------------------------------------------------------------
STEP 1: User Selects Upload Flow
--------------------------------------------------------------------------------
REQUEST PAYLOAD:
{
  "message": "{\"value\":\"UPLOAD\"}"
}

EXPECTED RESPONSE:
{
  "data": {
    "action": "ASK_UPLOAD_DOCUMENT",
    "message": "Please upload your medical document (Prescription, Lab Report, etc.).",
    "onboardingState": {
      "flowMode": "UPLOAD",
      "currentStep": "ASK_UPLOAD_DOCUMENT"
    }
  },
  "status": {
    "description": "Success",
    "status": "SUCCESS",
    "statusCode": 200
  }
}

--------------------------------------------------------------------------------
STEP 2: Document Processing Fails (Trigger Failure)
--------------------------------------------------------------------------------
(Simulate upload/OCR failure from client)

REQUEST PAYLOAD:
{
  "message": "OCR_FAILED"
}

EXPECTED RESPONSE:
{
  "data": {
    "action": "ASK_UPLOAD_DOCUMENT_FAILED",
    "message": "Document processing failed. Please try again or enter details manually.",
    "options": [
      {
        "label": "Retry Upload",
        "value": "RETRY_UPLOAD"
      },
      {
        "label": "Enter Details Manually",
        "value": "MANUAL"
      }
    ],
    "onboardingState": {
      "flowMode": "UPLOAD",
      "ocrFailed": true,
      "currentStep": "ASK_UPLOAD_DOCUMENT_FAILED"
    }
  },
  "status": {
    "description": "Success",
    "status": "SUCCESS",
    "statusCode": 200
  }
}
